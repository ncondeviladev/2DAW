1. Comandos de Sistema y Red (Linux / Docker)
Gestión de paquetes y edición de ficheros
apt update
# Actualizar lista de paquetes en contenedor
apt install -y <paquete>
# Instalar paquetes (ej. apache2, nginx, nano, curl)
nano <ruta_fichero>
# Edición de texto (Ctrl+O Guardar, Enter, Ctrl+X Salir)
Gestión de servicios (service)
service <servicio> status
# Consultar estado de ejecución (ej. apache2, nginx)
service <servicio> start|stop
# Iniciar o detener proceso del servicio
service <servicio> restart
# Reinicio completo para aplicar cambios de configuración
service <servicio> reload
# Recargar configuración sin desconectar proceso activo
Permisos, rutas y pruebas de red
chown -R www-data:www-data <ruta>
# Asignación de usuario web (-R: recursivo) (ej. /var/www/html)
chmod -R 755 <ruta>
# Permisos de lectura y ejecución para servidor web
ip a
# Consultar dirección IP asignada a la interfaz local
curl http://localhost:<puerto>
# Ver respuesta HTTP (En Windows: curl.exe | En Linux: curl)
curl -I http://localhost:<puerto>
# Inspección de cabeceras HTTP (comprobar código 200 OK)
curl -I http://localhost:<puerto>/<ruta_err>
# Verificación de código 404 Not Found personalizado
ssh <usuario>@<ip> [-p <puerto>]
# Conexión remota por SSH (-p: puerto si no es 22)
3. Servidor Web Apache2 (Sintaxis y Comandos)
Configuración de VirtualHosts y Puertos (PEC 1)
mkdir -p /var/www/<sitio> && nano /var/www/<sitio>/index.html
nano /var/www/<sitio>/404.html
# Estructura de ficheros web y error 404 personalizado

nano /etc/apache2/ports.conf
# Puerto de escucha general: Listen <puerto> (ej. Listen 8080)

# VirtualHost en /etc/apache2/sites-available/<sitio>.conf:
<VirtualHost *:<puerto>>
    DocumentRoot /var/www/<sitio>
    ServerName <dominio>                 # (ej. apache1.daw)
    ErrorDocument 404 /404.html
</VirtualHost>

a2ensite <sitio>.conf && a2dissite 000-default.conf
# Habilitar VirtualHost y deshabilitar sitio por defecto
apachectl configtest && service apache2 restart
# Comprobación de sintaxis (Syntax OK) y reinicio
Multisitio por Dominio en Apache
nano /etc/hosts
# Resolución local: 127.0.0.1 apache1.daw apache2.daw sitio1.daw
a2ensite <sitio1>.conf <sitio2>.conf
# Activación concurrente de VirtualHosts en el mismo puerto
2. Docker CLI & Redes (PEC 2)
Construcción, ejecución y ciclo de vida
docker build -t <nombre_imagen>:<tag> .
# Construir imagen (-t: tag | . : contexto carpeta actual)
docker run -d -p <p_host>:<p_cont> --name <nom> <img>
# Ejecución en segundo plano (-d) con mapeo de puertos
docker run -it -p <p1>:<p1> --name <nom> ubuntu bash
# Ejecución interactiva (-it) con terminal adjunta
docker ps / docker ps -a
# Listar contenedores activos / todos los estados
docker exec -it <contenedor> bash|sh
# Terminal interactiva interior (Alpine: sh | Ubuntu: bash)
docker start -ai <contenedor>
# Encender y adjuntar terminal si el contenedor estaba parado
docker logs [-f] <contenedor>
# Inspección de registros (-f: en tiempo real)
docker stop <cont> && docker rm [-f] <cont>
# Parada y eliminación de contenedor (-f: forzar borrado)
docker rmi <imagen>
# Eliminación de imagen local
Flags frecuentes explicados
-d: segundo plano | -it: interactivo/tty | -p: mapeo host:cont
-t: tag/nombre imagen | . : dir. actual | -v: volumen host:cont
--name: nombre cont. | -f: forzar/seguir logs | -ai: adjuntar tty
Multicontenedor en red bridge y volumen (PEC 2 Ej. 2)
docker network create <red> && docker volume create <vol>
# Creación de red bridge interna y volumen nombrado de datos
docker run -d --name <db> --network <red> -v <vol>:/var/lib/mysql -e MYSQL_ROOT_PASSWORD=<pass> mysql:8.0
# Despliegue de servicio BD persistente en red aislada
docker run -d --name <app> --network <red> -p <p_host>:80 -e PMA_HOST=<db> phpmyadmin/phpmyadmin
# Despliegue de aplicación web enlazada por nombre de host interno
docker exec -it <app> ping -c 3 <db>
# Comprobación de red (instalar si falta: apt update && apt install -y iputils-ping)
4. Servidor Web Nginx (Sintaxis y Comandos)
Configuración de Server Blocks y Puertos (PEC 1)
mkdir -p /var/www/<sitio> && nano /var/www/<sitio>/index.html
nano /var/www/<sitio>/404.html
# Creación de contenido web y página 404 personalizada

# Server Block en /etc/nginx/sites-available/<sitio>:
server {
    listen <puerto>;                    # (ej. 80 o 8080)
    root /var/www/<sitio>;
    index index.html index.htm;
    server_name <dominio>;               # (ej. sitio1.daw)
    error_page 404 /404.html;
    location = /404.html { internal; }
}

ln -s /etc/nginx/sites-available/<sitio> /etc/nginx/sites-enabled/
# Enlace simbólico (OBLIGATORIO usar ruta absoluta con / inicial)
rm /etc/nginx/sites-enabled/default
# Eliminación del bloque default para evitar conflictos de puerto
nginx -t && service nginx restart
# Test de configuración (test successful) y reinicio
Multisitio por Dominio en Nginx
ln -s /etc/nginx/sites-available/<sitio2> /etc/nginx/sites-enabled/
# Concurrencia de múltiples dominios en un mismo puerto (listen 80;)

5. Estructura de Dockerfile y Despliegue (PEC 2)
Estructura Dockerfile Unificada (Apache / Nginx sobre Alpine)
FROM httpd:alpine
# Imagen base: httpd:alpine (Apache) o nginx:alpine (Nginx)

COPY <origen_host> /usr/local/apache2/htdocs/
# Destino Apache: /usr/local/apache2/htdocs/
# Destino Nginx:  /usr/share/nginx/html/ (o con WORKDIR y COPY <origen> .)

EXPOSE 80
# Puerto interno documentado (80)
Construcción y arranque de imagen propia
docker build -t <nombre_imagen>:<tag> .
# Construir imagen (-t: tag/nombre | . : contexto carpeta actual)
docker run -d -p <p_host>:80 --name <nom> <nombre_imagen>:<tag>
# Despliegue con redirección de puerto al host (ej. -p 9090:80)
curl.exe http://localhost:<p_host>
# Comprobación de acceso web desde PowerShell
6. Docker Compose (Multicontenedor — PEC 3)
Comandos principales y paso de verificación MySQL
docker compose up -d
# Despliegue de la pila en segundo plano
docker exec -it <app> docker-php-ext-install mysqli && docker restart <app>
# Instalación y activación del controlador MySQL en PHP oficial
docker compose ps
# Comprobación de estado 'Up' y puertos expuestos
docker compose logs -f [<servicio>]
# Seguimiento de registros de salida de los servicios
docker compose down -v
# Detención del stack con eliminación de volúmenes (-v)
Estructura compose.yml (App + DB + phpMyAdmin)
services:
  <app>:
    image: php:8.3-apache
    container_name: <nom_app>
    ports:
      - "<p_host>:80"                  # (ej. "8000:80")
    volumes:
      # BIND MOUNT: Carpeta del host montada en contenedor
      - <ruta_host>:<ruta_cont>        # (ej. ./web:/var/www/html)
    depends_on:
      - <db>                           # Declaración en array (- db)
    restart: always
    environment:
      MYSQL_HOST: <db>                 # Resolución por nombre de servicio
      MYSQL_DATABASE: <nombre_bd>
      MYSQL_USER: <usuario>
      MYSQL_PASSWORD: <pass_user>

  <db>:
    image: mysql:8.0
    container_name: <nom_db>
    restart: always
    environment:
      MYSQL_ROOT_PASSWORD: <pass_root>
      MYSQL_DATABASE: <nombre_bd>
      MYSQL_USER: <usuario>
      MYSQL_PASSWORD: <pass_user>
    volumes:
      # NAMED VOLUME: Almacenamiento gestionado por Docker
      - <vol_datos>:/var/lib/mysql

  phpmyadmin:
    image: phpmyadmin/phpmyadmin
    container_name: phpmyadmin
    environment:
      PMA_HOST: <db>                   # Nombre del host de la base de datos
      PMA_USER: root
      PMA_PASSWORD: <pass_root>
    ports:
      - "<p_host>:80"                  # (ej. "8081:80")
    depends_on:
      - <db>
    restart: always

volumes:
  <vol_datos>:                         # Registro obligatorio de Named Volume (: final)
7. Kubernetes (kubectl & Manifiestos — PEC 3)
Comandos de gestión con kubectl
kubectl apply -f <fichero>.yml
# Aplicar configuración declarativa (ej. kube.yml)
kubectl get pods
# Consultar estado de los pods (Running, Pending, etc.)
kubectl get services
# Consultar servicios expuestos y puertos asignados
kubectl describe pod <nombre_pod>
# Diagnóstico y registro de eventos del pod
kubectl logs <nombre_pod>
# Registro de salida del contenedor en el pod
kubectl delete -f <fichero>.yml
# Eliminación de recursos definidos en el manifiesto
Estructura del manifiesto kube.yml (Deployment + Service)
apiVersion: apps/v1
kind: Deployment
metadata:
  name: <nombre_despliegue>            # (ej. web-deployment)
spec:
  replicas: <numero_replicas>          # (ej. 3)
  selector:
    matchLabels:
      app: <etiqueta_app>              # (ej. web)
  template:
    metadata:
      labels:
        app: <etiqueta_app>
    spec:
      containers:
      - name: <nombre_contenedor>      # (ej. nginx)
        image: <imagen>:<tag>          # (ej. nginx:alpine)
        ports:
        - containerPort: <puerto_interno> # (ej. 80)
---
apiVersion: v1
kind: Service
metadata:
  name: <nombre_servicio>              # (ej. web-service)
spec:
  type: NodePort                       # Exposición a través de puerto del nodo
  selector:
    app: <etiqueta_app>
  ports:
    - port: <puerto_servicio>          # (ej. 80)
      targetPort: <puerto_interno>     # (ej. 80)
      nodePort: <puerto_30000_32767>   # (ej. 30080)
