1. Comandos de Sistema y Red (Linux)
Gestión de paquetes y edición de ficheros
apt update
# Actualizar repositorio de paquetes
apt install -y <paquete>
# Instalar (-y: auto-aceptar) (ej. apache2, nginx, nano, curl)
nano <ruta_fichero>
# Editor de texto en terminal (Ctrl+O Guardar, Ctrl+X Salir)
Gestión de servicios (service)
service <servicio> status
# Consultar estado del servicio (ej. apache2, nginx)
service <servicio> start|stop
# Iniciar / Detener servicio (ej. apache2, nginx)
service <servicio> restart
# Reinicio completo (recomendado en Docker para asegurar arranque)
service <servicio> reload
# Recargar config sin corte (solo si el servicio ya estaba activo)
Permisos, rutas y pruebas de red
chown -R www-data:www-data <ruta>
# Usuario web de Apache/Nginx (-R: recursivo) (ej. /var/www/html)
chmod -R 755 <ruta>
# Permisos de lectura y ejecución (-R: recursivo a subcarpetas)
ip a
# Consultar IP asignada al contenedor o máquina
curl http://localhost:<puerto>
# Ver HTML (En Windows PowerShell: curl.exe | En Linux: curl)
curl -I http://localhost:<puerto>
# Solo cabeceras HTTP (-I: comprobar código 200 OK sin cuerpo)
curl -I http://localhost:<puerto>/<ruta_err>
# Probar ruta inexistente (-I: comprobar código 404 Not Found)
ssh <usuario>@<ip> [-p <puerto>]
# Acceso remoto por SSH (-p: puerto remoto si no es 22)
2. Servidor Web Apache2 (Sintaxis y Comandos)
Configuración de VirtualHosts y Puertos (PEC 1)
# Ficheros web y error 404 personalizado:
mkdir -p /var/www/<sitio> && nano /var/www/<sitio>/index.html
nano /var/www/<sitio>/404.html

# Puerto de escucha general (Listen):
nano /etc/apache2/ports.conf
# Añadir o cambiar directiva: Listen <puerto> (ej. 8080)

# VirtualHost en /etc/apache2/sites-available/<sitio>.conf:
<VirtualHost *:<puerto>>
    DocumentRoot /var/www/<sitio>
    ServerName <dominio_opcional>       # (ej. sitio1.daw)
    ErrorDocument 404 /404.html
</VirtualHost>

# Habilitación de sitios (a2ensite / a2dissite):
a2ensite <sitio>.conf && a2dissite 000-default.conf

# Validación de sintaxis y reinicio:
apachectl configtest
# Debe responder: Syntax OK
service apache2 restart
# Aplicar cambios en Apache
Multisitio en Apache (≥2 sitios simultáneos)
# Por Dominio: ServerName distinto en cada .conf (ej. sitio1.daw / sitio2.daw)
nano /etc/hosts
# Añadir resolución local: 127.0.0.1 <dominio1> <dominio2>
# Por Puerto: En ports.conf Listen 8080 y Listen 8081 (un puerto por VirtualHost)
a2ensite <sitio1>.conf <sitio2>.conf
# Activar ambos sitios a la vez
3. Servidor Web Nginx (Sintaxis y Comandos)
Configuración de Server Blocks y Puertos (PEC 1)
# Ficheros web y error 404 personalizado:
mkdir -p /var/www/<sitio> && nano /var/www/<sitio>/index.html
nano /var/www/<sitio>/404.html

# Server Block en /etc/nginx/sites-available/<sitio>:
server {
    listen <puerto>;                    # (ej. 80, 8989)
    root /var/www/<sitio>;
    index index.html index.htm;
    server_name <dominio_o__>;          # (ej. _ o sitio2.daw)
    error_page 404 /404.html;
    location = /404.html { internal; }
}

# Enlace simbólico (OBLIGATORIO usar ruta absoluta con / inicial):
ln -s /etc/nginx/sites-available/<sitio> /etc/nginx/sites-enabled/

# Desactivar sitio por defecto para evitar conflictos en puerto 80:
rm /etc/nginx/sites-enabled/default

# Validación de sintaxis y arranque:
nginx -t
# Debe responder: syntax is ok / test is successful
service nginx restart
# Reiniciar servicio Nginx (o service nginx start si estaba parado)
Multisitio en Nginx (≥2 sitios simultáneos)
# Por Dominio: Mismo puerto (80) y server_name distinto (ej. sitio1.daw / sitio2.daw)
nano /etc/hosts
# Añadir resolución local: 127.0.0.1 <dominio1> <dominio2>
# Por Puerto: Diferente puerto (listen 80; y listen 8989;) en cada bloque
ln -s /etc/nginx/sites-available/<sitio2> /etc/nginx/sites-enabled/
# Activar segundo sitio con su enlace simbólico
4. Docker CLI (Contenedores y Administración)
Ejecución interactiva y en segundo plano
docker run -d -p <p_host>:<p_cont> --name <nom> <img>
# -d: segundo plano | -p: puertos host:cont | --name: nombre
docker run -it -p <p1>:<p1> -p <p2>:<p2> --name <nom> ubuntu bash
# -it: terminal interactiva | Contenedor de pruebas con puertos
Flags frecuentes explicados
-d: segundo plano | -it: interactivo/tty | -p: mapeo host:cont
-t: tag/nombre imagen | . : directorio actual | -v: volumen host:cont
--name: nombre cont. | -f: forzar/seguir logs | -ai: adjuntar tty
Ciclo de vida y depuración
docker ps / docker ps -a
# Listar contenedores activos / -a: todos (incluso parados)
docker exec -it <contenedor> bash|sh
# -it: ENTRAR a consola si YA está encendido (En Alpine: sh)
docker exec -it <cont> docker-php-ext-install mysqli
# Instalar driver MySQL en PHP oficial (soluciona 'mysqli not found')
docker start -ai <contenedor>
# -ai: ENCENDER y entrar si estaba parado
docker logs [-f] <contenedor>
# Ver registros de salida (-f: seguir en tiempo real)
docker stop <cont> && docker rm [-f] <cont>
# Detener y eliminar contenedor (-f: forzar borrado activo)
docker rmi <imagen>
# Borrar imagen del sistema local
Flujo Red + Volumen + Enlace CLI (PEC 2 Ej. 2)
docker network create <red> && docker volume create <vol>
# 1. Crear red bridge interna y volumen nombrado persistente
docker run -d --name <db> --network <red> -v <vol>:/var/lib/mysql -e MYSQL_ROOT_PASSWORD=<pass> mysql:8.0
# 2. Arrancar BD conectada a la red y con volumen de datos
docker run -d --name <app> --network <red> -p <p_host>:80 -e PMA_HOST=<db> phpmyadmin/phpmyadmin
# 3. Arrancar app web en la misma red enlazada a BD por nombre
docker exec -it <app> ping -c 3 <db>
# 4. Probar conectividad interna (instalar si falta: apt update && apt install -y iputils-ping)

5. Estructura de Dockerfile y Despliegue (PEC 2)
Ejemplo Apache Alpine (PEC 2 - Octopus)
FROM httpd:alpine
COPY <origen_host> /usr/local/apache2/htdocs/
# Origen en tu PC (ej. static/ o index.html) -> Destino en contenedor
EXPOSE 80                              # Puerto interno informativo
Ejemplo Nginx Alpine (Directo o con WORKDIR)
FROM nginx:alpine
# Opción A (directa): COPY <origen_host> /usr/share/nginx/html/
# Opción B (con WORKDIR): WORKDIR /usr/share/nginx/html && COPY <origen> .
# El punto (.) en COPY indica el directorio de trabajo interior
EXPOSE 80                              # Puerto interno informativo
Construcción y arranque del contenedor (Flujo PEC 2)
docker build -t <nombre_imagen>:<tag> .
# -t: tag/nombre (ej. examen-web:1.0) | . : directorio actual con Dockerfile
docker run -d -p <p_host>:80 --name <nom> <nombre_imagen>:<tag>
# Arrancar (-d: segundo plano | -p: puerto externo:interno | --name: nombre)
docker logs [-f] <nom> && curl.exe http://localhost:<p_host>
# -f: seguir logs | Probar web (En Windows: curl.exe | En Linux: curl)
6. Docker Compose (Multicontenedor)
Ciclo de vida y comandos principales
docker compose up -d
# Levantar stack (-d: segundo plano sin bloquear terminal)
docker compose ps
# Ver estado y puertos de los contenedores del stack
docker compose logs -f [<servicio>]
# Ver registros (-f: seguir salida en tiempo real)
docker compose down
# Detener y eliminar contenedores y redes del stack
docker compose down -v
# Detener y borrar también volúmenes con persistencia (-v)
Estructura real compose.yml (PEC 3: App + DB + PhpMyAdmin)
services:
  <servicio_app>:                      # (ej. app, web)
    image: <imagen>:<tag>              # (ej. php:8.3-apache)
    container_name: <nombre_cont>      # (ej. php_app)
    ports:
      - "<p_host>:<p_cont>"            # (ej. "8080:80")
    volumes:
      - <ruta_host>:<ruta_cont>        # Bind mount: carpeta PC (ej. ./web:/var/www/html)
    depends_on:
      - <servicio_db>                  # (ej. db)
    restart: always
    environment:
      MYSQL_HOST: <servicio_db>        # Host = nombre servicio (ej. db)
      MYSQL_DATABASE: <nombre_bd>      # (ej. examen_db)
      MYSQL_USER: <usuario>            # (ej. examen_user)
      MYSQL_PASSWORD: <pass_user>      # (ej. examen_pass)

  <servicio_db>:                       # (ej. db)
    image: mysql:8.0
    container_name: <nombre_cont>      # (ej. blog)
    restart: always
    environment:
      MYSQL_ROOT_PASSWORD: <pass_root> # (ej. rootpass)
      MYSQL_DATABASE: <nombre_bd>      # (ej. examen_db)
      MYSQL_USER: <usuario>            # (ej. examen_user)
      MYSQL_PASSWORD: <pass_user>      # (ej. examen_pass)
    volumes:
      - <vol_datos>:/var/lib/mysql     # Named volume: Docker (ej. datos_examen:/var/lib/mysql)

  phpmyadmin:
    image: phpmyadmin/phpmyadmin
    container_name: phpmyadmin
    environment:
      PMA_HOST: <servicio_db>          # (ej. db)
      PMA_USER: root
      PMA_PASSWORD: <pass>
    ports:
      - "<p_host>:80"                  # (ej. "8081:80")
    depends_on:
      - <servicio_db>                  # (ej. db)
    restart: always

volumes:
  <vol_datos>:                         # Registro Named Volume (: obligatorio)
7. Kubernetes (kubectl & Manifiestos)
Comandos de gestión con kubectl
kubectl apply -f <fichero>.yml
# Aplicar despliegue o servicio (ej. kube.yml)
kubectl get pods
# Listar estado de los pods (Running, Pending, CrashLoopBackOff)
kubectl get services
# Listar servicios activos y sus puertos mapeados
kubectl describe pod <nombre_pod>
# Información detallada y eventos de depuración del pod
kubectl logs <nombre_pod>
# Ver la salida por consola del contenedor dentro del pod
kubectl delete -f <fichero>.yml
# Eliminar despliegue y recursos creados
Estructura del manifiesto kube.yml (Deployment + Service)
apiVersion: apps/v1
kind: Deployment
metadata:
  name: <nombre_despliegue>            # (ej. web-deployment)
spec:
  replicas: <numero_replicas>          # (ej. 3)
  selector:
    matchLabels:
      app: <etiqueta_app>              # (ej. web)
  template:
    metadata:
      labels:
        app: <etiqueta_app>
    spec:
      containers:
      - name: <nombre_contenedor>      # (ej. nginx)
        image: <imagen>:<tag>          # (ej. nginx:alpine)
        ports:
        - containerPort: <puerto_interno> # (ej. 80)
---
apiVersion: v1
kind: Service
metadata:
  name: <nombre_servicio>              # (ej. web-service)
spec:
  type: NodePort                       # NodePort para acceso exterior
  selector:
    app: <etiqueta_app>
  ports:
    - port: <puerto_servicio>          # (ej. 80)
      targetPort: <puerto_interno>     # (ej. 80)
      nodePort: <puerto_30000_32767>   # (ej. 30080)
