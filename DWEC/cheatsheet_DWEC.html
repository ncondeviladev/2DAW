class Producto {
    // 1. Campos privados nativos obligatorios declarados arriba con #:
    #id;
    #nombre;
    #precio;
    static totalCreados = 0; // Propiedad estática de clase

    // 2. Constructor:
    constructor(id, nombre, precio) {
        this.#id = id;
        this.nombre = nombre; // Invoca al setter para sanear
        this.precio = precio; // Invoca al setter para validar
        Producto.totalCreados++;
    }

    // 3. Getters y Setters con validaciones de tipo y rango:
    get id() { return this.#id; } // Solo lectura (sin setter)

    get nombre() { return this.#nombre; }
    set nombre(val) {
        if (!val || typeof val !== "string") {
            throw new Error("Nombre no válido");
        }
        this.#nombre = val.trim();
    }

    get precio() { return this.#precio; }
    set precio(val) {
        if (typeof val !== "number" || val < 0) {
            throw new RangeError("El precio debe ser >= 0");
        }
        this.#precio = val;
    }

    // 4. Métodos de negocio y método resumen:
    aplicarDescuento(porcentaje) {
        this.#precio -= (this.#precio * porcentaje) / 100;
    }

    resumen() {
        return `[${this.#id}] ${this.#nombre} - ${this.#precio}€`;
    }

    // 5. Método estático de clase:
    static mostrarTotal() {
        return `Total productos creados: ${Producto.totalCreados}`;
    }
}

Plantilla ejRA2.js (String + Math + Date)
function procesarRA2(cadena) {
    // 1. Limpieza y división de campos:
    const partes = cadena.trim().split("/");
    
    // 2. Tratamiento de texto (capitalizar y rellenar ceros):
    const codigo = partes[0].padStart(5, "0");
    const nombre = partes[1].trim().toUpperCase();
    
    // 3. Manejo de fechas y cálculo de días restantes:
    const fObjetivo = new Date(partes[2]);
    const hoy = new Date();
    hoy.setHours(0, 0, 0, 0);
    fObjetivo.setHours(0, 0, 0, 0);
    const dias = Math.max(0, Math.round((fObjetivo - hoy) / (1000*60*60*24)));
    
    return { codigo, nombre, dias };
}
Plantilla ejRA3.js (filter + map + reduce + sort)
function procesarRA3(items) {
    if (!Array.isArray(items) || items.length === 0) {
        return { activos: [], total: 0, todosValidos: false };
    }
    // 1. Filter + Map + Sort en cadena (artículos con stock):
    const activos = items
        .filter(x => x.unidades > 0)
        .map(x => x.ref)
        .sort((a, b) => a.localeCompare(b));
        
    // 2. Reduce: sumatorio económico total:
    const total = parseFloat(items.reduce((acc, x) => acc + (x.precio * x.unidades), 0).toFixed(2));
    
    // 3. Every / Some: validaciones booleanas:
    const todosValidos = items.every(x => x.precio > 0);
    
    return { activos, total, todosValidos };
}

class Elemento {
    #id;
    #minutos;
    constructor(id, titulo, minutos) {
        this.#id = id;
        this.titulo = titulo.trim();
        this.completada = false;
        this.minutos = minutos; // Llama al setter
    }
    get id() { return this.#id; }
    get minutos() { return this.#minutos; }
    set minutos(val) {
        this.#minutos = (typeof val === "number" && val > 0) ? val : 30;
    }
    completar() { this.completada = true; }
    resumen() {
        const est = this.completada ? "COMPLETADA" : "PENDIENTE";
        return `[ID: ${this.#id}] ${this.titulo} | ${est} | ${this.#minutos} min`;
    }
}
