5. SELECTORES, AGRUPADORES Y PESOS (S05-S06)
/* SELECTORES BÁSICOS: */
* { }           /* UNIVERSAL */
div { }         /* ELEMENTO / ETIQUETA */
.clase { }      /* CLASE */
#ident { }      /* ID */
/* AGRUPADORES: */
h1, h2, h3 { }  /* COMA: AGRUPACIÓN MÚLTIPLE */
div p { }       /* ESPACIO: HIJO CUALQUIER NIVEL */
div > p { }     /* MAYOR: HIJO DIRECTO 1ER NIVEL */
h2 + p { }      /* MÁS: HERMANO ADYACENTE */
h2 ~ p { }      /* VIRGULILLA: HERMANO GENERAL */
/* SELECTORES POR ATRIBUTO: */
input[type="text"] { }   /* VALOR EXACTO */
input[name^="txt"] { }   /* EMPIEZA POR "txt" */
input[name$="id"] { }    /* TERMINA POR "id" */
input[name*="val"] { }   /* CONTIENE "val" */
/* PSEUDO-CLASES Y PSEUDO-ELEMENTOS: */
:root { }  a:hover { }  a:active { }
input:focus { }
li:first-child { }  li:last-child { }
tr:nth-child(even) { }   /* FILAS PARES */
tr:nth-child(odd) { }    /* FILAS IMPARES */
h1::before { content: "★ "; }
h1::after { content: " ↗"; }
:not(.activo) { }  :is(h1,h2) { }  :has(img) { }
